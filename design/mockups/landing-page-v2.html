Built for developers.

Integrate our robust API in seconds. We handle the deliverability, bounce tracking, and compliance automatically, so you can focus on your product.

send-welcome.ts package.json 
import { SouraMail } from '@souramail/node';
import { WelcomeTemplate } from './emails/welcome';

const soura = new SouraMail(process.env.SOURA_KEY);

export async function sendWelcomeEmail(user) {
  // Send via global edge network with automatic failover
  const response = await soura.emails.send({
    from: 'Acme Corp <hello@acme.com>',
    to: user.email,
    subject: 'Welcome to the future of email',
    react: WelcomeTemplate({ firstName: user.name }),
    tags: ['onboarding', 'welcome'],
    trackOpens: true
  });

  return response.id;
}
